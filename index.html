    <button onclick="comecar()">Jogar</button>

  <div id="fim" style="display:none;">

    <h2>Fim de Jogo!</h2>

    <p>Pontuação: <span id="pontosFim">0</span></p>

    <button onclick="comecar()">Jogar de Novo</button>

    <button class="btn-secundario" onclick="voltarInicio()">Voltar ao Início</button>

<script>

  // ====== CONFIGURAÇÃO DO TELEGRAM WEB APP ======

  const tg = window.Telegram?.WebApp;

  if (tg) {

    tg.ready();   // Avisa o Telegram que o app carregou

    tg.expand();  // Faz o app ocupar a tela inteira

  }


  // ====== VARIÁVEIS DO JOGO ======

  let pontos = 0, tempo = 30, timer, jogando = false;

  const alvo = document.getElementById('alvo');

  const elTempo = document.getElementById('tempo');

  const elPontos = document.getElementById('pontos');

  const inicio = document.getElementById('inicio');

  const fim = document.getElementById('fim');


  // ====== LÓGICA DO JOGO ======

  function moverAlvo() {

    // Calcula posição aleatória, evitando o topo (HUD) e as bordas

    const margemTopo = 80; // Evita o placar

    const tamanhoAlvo = 80;

    

    const maxX = window.innerWidth - tamanhoAlvo;

    const maxY = window.innerHeight - tamanhoAlvo - margemTopo;

    

    alvo.style.left = Math.random() * maxX + 'px';

    alvo.style.top = (Math.random() * maxY + margemTopo) + 'px';

  }


  function comecar() {

    pontos = 0; 

    tempo = 30; 

    jogando = true;

    

    elPontos.textContent = 0; 

    elTempo.textContent = 30;

    

    inicio.style.display = 'none'; 

    fim.style.display = 'none';

    alvo.style.display = 'block'; 

    

    moverAlvo();


    clearInterval(timer);

    timer = setInterval(() => {

      tempo--; 

      elTempo.textContent = tempo;

      if (tempo <= 0) encerrar();

    }, 1000);

  }


  function encerrar() {

    jogando = false; 

    clearInterval(timer);

    alvo.style.display = 'none';

    

    document.getElementById('pontosFim').textContent = pontos;

    fim.style.display = 'block';


    // Envia a pontuação para o bot do Telegram (opcional)

    if (tg) {

      tg.sendData(JSON.stringify({ pontuacao: pontos }));

    }

  }


  function voltarInicio() {

    fim.style.display = 'none';

    inicio.style.display = 'block';

  }


  // Evento de clique no alvo

  alvo.addEventListener('click', (e) => {

    e.preventDefault(); // Evita zoom/scroll no mobile

    if (!jogando) return;

    pontos++; 

    elPontos.textContent = pontos; 

    moverAlvo();

  });


  // Previne o comportamento padrão de arrastar na tela

  document.addEventListener('touchmove', (e) => {

    if (jogando) e.preventDefault();

  }, { passive: false });

</script>
